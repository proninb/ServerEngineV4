cmake_minimum_required(VERSION 3.25)
project(StringPathReuse10C13 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PLS10_CORE_LIBRARY "" CACHE FILEPATH "Matching C10 Release IPO core")
if(NOT PLS10_CORE_LIBRARY)
 message(FATAL_ERROR "Set PLS10_CORE_LIBRARY")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(PublishStringReuse10C13 "${CMAKE_CURRENT_BINARY_DIR}/generated/native/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/native/publish_parser.cpp")
add_executable(StringReuseAudit10C13 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_graph.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_identity.cpp" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c10/allocation.cpp")
foreach(target PublishStringReuse10C13 StringReuseAudit10C13)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
 target_compile_definitions(${target} PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
 target_compile_options(${target} PRIVATE /W4 /permissive- /O2)
 set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
 target_link_libraries(${target} PRIVATE "${PLS10_CORE_LIBRARY}" psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
endforeach()
target_include_directories(PublishStringReuse10C13 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/native")
target_include_directories(StringReuseAudit10C13 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(StringReuseAudit10C13 PRIVATE PLS10_C9_AUDIT=1 PLS13_AUDIT=1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py reuse.hpp)
enable_testing()
add_test(NAME PLS10C13_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py" "$<TARGET_FILE:PublishStringReuse10C13>" "$<TARGET_FILE:StringReuseAudit10C13>" "${ENGINE_ROOT}/build/parallel-local-source-10-c10/Release/PublishScratch10C10.exe" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
