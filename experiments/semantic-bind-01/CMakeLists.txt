cmake_minimum_required(VERSION 3.25)
project(SemanticBind01 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SERVER_ENGINE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ENGINE_ROOT}" engine)
get_target_property(engine_sources ServerEngineV4 SOURCES)
list(FILTER engine_sources EXCLUDE REGEX "server_engine/main\\.cpp$")
set(full_sources)
foreach(src IN LISTS engine_sources)
  if(IS_ABSOLUTE "${src}")
    list(APPEND full_sources "${src}")
  else()
    list(APPEND full_sources "${ENGINE_ROOT}/${src}")
  endif()
endforeach()
add_executable(SemanticBind01 main.cpp ${full_sources})
target_include_directories(SemanticBind01 PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
if(WIN32)
  target_link_libraries(SemanticBind01 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(SemanticBind01 PRIVATE Threads::Threads)
endif()
if(MSVC)
  target_compile_options(SemanticBind01 PRIVATE /W4 /permissive- /O2)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_ok)
if(ipo_ok)
  set_property(TARGET SemanticBind01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
enable_testing()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_test(NAME semantic_bind_model COMMAND SemanticBind01 --self-test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME semantic_bind_oracle COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/test_oracle.py" "$<TARGET_FILE:SemanticBind01>"
      "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
endif()
