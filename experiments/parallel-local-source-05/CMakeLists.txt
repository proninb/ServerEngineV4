cmake_minimum_required(VERSION 3.25)
project(ParallelLocalSource05 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(SERVER_ENGINE_BUILD_BENCHMARKS ON CACHE BOOL "Core sources" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Own tests" FORCE)
add_subdirectory("${ENGINE_ROOT}" engine EXCLUDE_FROM_ALL)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py audit.hpp
 "${ENGINE_ROOT}/experiments/parallel-local-source-04/generate.py"
 "${ENGINE_ROOT}/experiments/parallel-local-source-04/memory.hpp"
 "${ENGINE_ROOT}/experiments/parallel-local-source-03/generate.py"
 "${ENGINE_ROOT}/experiments/parallel-local-source-03/probe.hpp"
 "${ENGINE_ROOT}/experiments/parallel-local-source-03/flat.hpp"
 "${ENGINE_ROOT}/experiments/parallel-local-source-02/bind.hpp"
 "${ENGINE_ROOT}/experiments/parallel-local-source-02/main.cpp" "${ENGINE_ROOT}/experiments/parallel-local-source-02/aliases.hpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(sources ServerEngineV4PublishBenchmark SOURCES)
set(core)
foreach(source IN LISTS sources)
 if(NOT IS_ABSOLUTE "${source}")
  set(source "${ENGINE_ROOT}/${source}")
 endif()
 if(NOT source MATCHES "publish_benchmark[.]cpp$")
  list(APPEND core "${source}")
 endif()
endforeach()
add_library(PLS05Core STATIC ${core})
add_executable(ParallelLocalSource05 "${CMAKE_CURRENT_BINARY_DIR}/generated/control/main.cpp")
add_executable(ParallelLocalSource05Audit "${CMAKE_CURRENT_BINARY_DIR}/generated/audit/main.cpp")
get_directory_property(definitions DIRECTORY "${ENGINE_ROOT}" COMPILE_DEFINITIONS)
foreach(target PLS05Core ParallelLocalSource05 ParallelLocalSource05Audit)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine")
 target_compile_definitions(${target} PRIVATE ${definitions})
 foreach(property COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE)
  get_target_property(value ServerEngineV4PublishBenchmark ${property})
  if(value AND NOT value MATCHES "NOTFOUND$")
   set_property(TARGET ${target} PROPERTY ${property} "${value}")
  endif()
 endforeach()
endforeach()
foreach(target ParallelLocalSource05 ParallelLocalSource05Audit)
 target_compile_definitions(${target} PRIVATE PLS03_PROBE=1)
 target_link_libraries(${target} PRIVATE PLS05Core psapi)
 foreach(property LINK_LIBRARIES LINK_OPTIONS)
  get_target_property(value ServerEngineV4PublishBenchmark ${property})
  if(value AND NOT value MATCHES "NOTFOUND$")
   set_property(TARGET ${target} APPEND PROPERTY ${property} "${value}")
  endif()
 endforeach()
endforeach()
target_compile_definitions(ParallelLocalSource05Audit PRIVATE PLS04_ALLOCATIONS=1)
add_custom_target(PLS05 DEPENDS ParallelLocalSource05 ParallelLocalSource05Audit)
enable_testing()
foreach(target ParallelLocalSource05 ParallelLocalSource05Audit)
 add_test(NAME ${target}_fixtures COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-02/test.py"
  "$<TARGET_FILE:${target}>" "${CMAKE_CURRENT_BINARY_DIR}/${target}-fixtures")
 set_tests_properties(${target}_fixtures PROPERTIES ENVIRONMENT "PLS03_MODE=1;PLS04_EARLY=1")
endforeach()

# Keep allocation observer outside cross-module optimization (MSVC 14.51 ICE).
if(MSVC)
 target_compile_options(ParallelLocalSource05Audit PRIVATE /GL-)
endif()
