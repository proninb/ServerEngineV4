cmake_minimum_required(VERSION 3.25)
project(SourceCriticalLinkReplay LANGUAGES CXX)
get_filename_component(REPO "../../.." ABSOLUTE)
set(SNAPSHOT "${REPO}/build/source-v2-critical-01")
file(GLOB OBJECTS "${SNAPSHOT}/build/ServerEngineV4PublishBenchmark.dir/Release/*.obj")
list(FILTER OBJECTS EXCLUDE REGEX "/publish_benchmark[.]obj$")
if(NOT OBJECTS)
 message(FATAL_ERROR "Build the frozen optimized production benchmark first")
endif()
set_source_files_properties(${OBJECTS} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(SourceCriticalLinkReplay main.cpp ${OBJECTS})
target_include_directories(SourceCriticalLinkReplay PRIVATE "${SNAPSHOT}/source/server_engine")
target_compile_features(SourceCriticalLinkReplay PRIVATE cxx_std_20)
target_compile_options(SourceCriticalLinkReplay PRIVATE /O2 /GL /Zi /Zo /W4 /permissive-)
set_property(TARGET SourceCriticalLinkReplay PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
target_link_options(SourceCriticalLinkReplay PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
target_link_libraries(SourceCriticalLinkReplay PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
