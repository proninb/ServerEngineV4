cmake_minimum_required(VERSION 3.25)
project(HeaderGraphIndex01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/header-v2-state-lazy-01/ServerEngineV4PublishBenchmark.dir/Release" CACHE PATH "Validated V2+lazy Release+IPO benchmark objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
list(FILTER engine_objects EXCLUDE REGEX "[/\\](graph|publish_benchmark)\\.obj$")
if(NOT engine_objects)
  message(FATAL_ERROR "Build the accepted V2+lazy benchmark first")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(HeaderGraphIndex01Audit "${ENGINE_ROOT}/benchmarks/publish_benchmark.cpp" "${ENGINE_ROOT}/server_engine/project/graph/graph.cpp" ${engine_objects})
target_compile_features(HeaderGraphIndex01Audit PRIVATE cxx_std_20)
target_compile_definitions(HeaderGraphIndex01Audit PRIVATE CW_GRAPH_MEMBER_INDEX_AUDIT=1 CW_HEADER_STATE_LAZY_AB=1 CW_GRAPH_MEMBER_INDEX_AB=1)
target_compile_options(HeaderGraphIndex01Audit PRIVATE /O2 /Ob2 /DNDEBUG /Zi /W4 /permissive-)
target_include_directories(HeaderGraphIndex01Audit PRIVATE "${ENGINE_ROOT}/server_engine")
target_link_options(HeaderGraphIndex01Audit PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
target_link_libraries(HeaderGraphIndex01Audit PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
set_property(TARGET HeaderGraphIndex01Audit PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_executable(HeaderGraphIndex01 "${ENGINE_ROOT}/benchmarks/publish_benchmark.cpp" "${ENGINE_ROOT}/server_engine/project/graph/graph.cpp" ${engine_objects})
target_compile_features(HeaderGraphIndex01 PRIVATE cxx_std_20)
target_compile_definitions(HeaderGraphIndex01 PRIVATE CW_HEADER_STATE_LAZY_AB=1 CW_GRAPH_MEMBER_INDEX_AB=1)
target_compile_options(HeaderGraphIndex01 PRIVATE /O2 /Ob2 /DNDEBUG /Zi /W4 /permissive-)
target_include_directories(HeaderGraphIndex01 PRIVATE "${ENGINE_ROOT}/server_engine")
target_link_options(HeaderGraphIndex01 PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
target_link_libraries(HeaderGraphIndex01 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
set_property(TARGET HeaderGraphIndex01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_executable(HeaderGraphIndex01Tests test.cpp "${ENGINE_ROOT}/server_engine/project/graph/graph.cpp" ${engine_objects})
target_compile_features(HeaderGraphIndex01Tests PRIVATE cxx_std_20)
target_compile_definitions(HeaderGraphIndex01Tests PRIVATE CW_HEADER_STATE_LAZY_AB=1 CW_GRAPH_MEMBER_INDEX_AB=1 CW_GRAPH_MEMBER_INDEX_TEST_FAILURE=1)
target_compile_options(HeaderGraphIndex01Tests PRIVATE /O2 /Ob2 /DNDEBUG /W4 /permissive-)
target_include_directories(HeaderGraphIndex01Tests PRIVATE "${ENGINE_ROOT}/server_engine")
target_link_libraries(HeaderGraphIndex01Tests PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
set_property(TARGET HeaderGraphIndex01Tests PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
enable_testing()
add_test(NAME header_graph_index_01 COMMAND HeaderGraphIndex01Tests)
add_executable(HeaderGraphIndex01Regression "${ENGINE_ROOT}/tests/compiled_project_tests.cpp" "${ENGINE_ROOT}/server_engine/project/graph/graph.cpp" ${engine_objects})
target_sources(HeaderGraphIndex01Regression PRIVATE "${ENGINE_ROOT}/server_engine/project/runtime/runtime_layout.cpp" "${ENGINE_ROOT}/server_engine/project/runtime/fixed_direct_materializer.cpp")
target_compile_features(HeaderGraphIndex01Regression PRIVATE cxx_std_20)
target_compile_definitions(HeaderGraphIndex01Regression PRIVATE CW_HEADER_STATE_LAZY_AB=1)
target_compile_options(HeaderGraphIndex01Regression PRIVATE /O2 /Ob2 /DNDEBUG /W4 /permissive-)
target_include_directories(HeaderGraphIndex01Regression PRIVATE "${ENGINE_ROOT}/server_engine")

set_property(TARGET HeaderGraphIndex01Regression PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME header_graph_index_01_compiled_project COMMAND HeaderGraphIndex01Regression)
set_tests_properties(header_graph_index_01 header_graph_index_01_compiled_project PROPERTIES RUN_SERIAL TRUE)
