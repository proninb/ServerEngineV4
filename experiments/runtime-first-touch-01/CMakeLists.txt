cmake_minimum_required(VERSION 3.25)
project(RuntimeFirstTouch01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/publish-critical-path-02/symbols/ServerEngineV4PublishBenchmark.dir/Release" CACHE PATH "Complete post-SHA production Release+IPO objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
list(FILTER engine_objects EXCLUDE REGEX "[/\\](project_runtime|publish_benchmark)\.obj$")
if(NOT engine_objects)
 message(FATAL_ERROR "Build the current complete production benchmark first")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(RuntimeFirstTouch01 "${ENGINE_ROOT}/benchmarks/publish_benchmark.cpp" "${ENGINE_ROOT}/server_engine/project/runtime/project_runtime.cpp" ${engine_objects})
add_executable(RuntimeFirstTouch01Tests "${ENGINE_ROOT}/tests/compiled_project_tests.cpp" "${ENGINE_ROOT}/server_engine/project/runtime/project_runtime.cpp" ${engine_objects})
target_sources(RuntimeFirstTouch01Tests PRIVATE "${ENGINE_ROOT}/server_engine/project/runtime/runtime_layout.cpp" "${ENGINE_ROOT}/server_engine/project/runtime/fixed_direct_materializer.cpp")
foreach(target RuntimeFirstTouch01 RuntimeFirstTouch01Tests)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_definitions(${target} PRIVATE CW_RUNTIME_OBJECT_LANES_AB=1 CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
 target_compile_options(${target} PRIVATE /O2 /Ob2 /DNDEBUG /Zi /W4 /permissive-)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine")
 target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
 target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
 set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endforeach()
enable_testing()
add_test(NAME runtime_first_touch_contract COMMAND RuntimeFirstTouch01Tests)
