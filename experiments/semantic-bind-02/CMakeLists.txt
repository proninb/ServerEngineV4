cmake_minimum_required(VERSION 3.25)
project(SemanticBind02 LANGUAGES CXX)
include(CheckIPOSupported)
check_ipo_supported(RESULT audit_ipo_ok)
add_subdirectory("../semantic-bind-01" sb01)
get_target_property(full_sources SemanticBind01 SOURCES)
list(FILTER full_sources EXCLUDE REGEX "(^|/)main\\.cpp$")
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
foreach(kind IN ITEMS Timing Allocations)
  set(target "SemanticBind02${kind}")
  add_executable(${target} audit.cpp ${full_sources})
  target_include_directories(${target} PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(WIN32)
    target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
  else()
    target_link_libraries(${target} PRIVATE Threads::Threads)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /O2)
  endif()
  if(audit_ipo_ok)
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
  if(kind STREQUAL "Allocations")
    target_compile_definitions(${target} PRIVATE SB02_ALLOCATIONS=1 SB02_REQUEST_LIFETIME=1)
  endif()
endforeach()
enable_testing()
add_test(NAME audit_timing_model COMMAND SemanticBind02Timing --self-test)
add_test(NAME audit_allocations_model COMMAND SemanticBind02Allocations --self-test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME audit_consistency COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/test_audit.py" "$<TARGET_FILE:SemanticBind02Timing>"
  "$<TARGET_FILE:SemanticBind02Allocations>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
