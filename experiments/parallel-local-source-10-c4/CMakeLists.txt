cmake_minimum_required(VERSION 3.25)
project(ProjectVisibleSource10C4 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-08" base EXCLUDE_FROM_ALL)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py stats.hpp "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3/publish_source.hpp" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3/commit.hpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(PublishProjectSource10C4 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp")
target_include_directories(PublishProjectSource10C4 PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
target_compile_features(PublishProjectSource10C4 PRIVATE cxx_std_20)
foreach(property LINK_LIBRARIES COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
 get_target_property(value PublishFinalState08 ${property})
 if(value AND NOT value MATCHES "NOTFOUND$")
  set_property(TARGET PublishProjectSource10C4 PROPERTY ${property} "${value}")
 endif()
endforeach()
enable_testing()
add_test(NAME PLS10C4_project_visibility COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py" "$<TARGET_FILE:PublishProjectSource10C4>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
