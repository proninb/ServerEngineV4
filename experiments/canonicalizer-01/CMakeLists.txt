cmake_minimum_required(VERSION 3.25)
project(Canonicalizer01 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(Canonicalizer01 main.cpp)
target_link_libraries(Canonicalizer01 PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(Canonicalizer01 PRIVATE Psapi)
endif()
if(MSVC)
  target_compile_options(Canonicalizer01 PRIVATE /W4 /permissive- /O2)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_ok)
if(ipo_ok)
  set_property(TARGET Canonicalizer01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
enable_testing()
add_test(NAME canonicalizer_model COMMAND Canonicalizer01 self-test)
add_executable(Canonicalizer01ModelTests model_tests.cpp)
target_link_libraries(Canonicalizer01ModelTests PRIVATE Threads::Threads)
add_test(NAME canonicalizer_collisions COMMAND Canonicalizer01ModelTests)
