cmake_minimum_required(VERSION 3.25)
project(StateLazy01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/semantic-bind-03/SemanticBind03Engine.dir/Release" CACHE PATH "Previously built MSVC Release engine objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
if(NOT engine_objects)
  message(FATAL_ERROR "Build SemanticBind03Engine first and pass its Release object directory")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(StateLazy01 main.cpp ${engine_objects})
target_compile_features(StateLazy01 PRIVATE cxx_std_20)
target_include_directories(StateLazy01 PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(StateLazy01 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
if(MSVC)
  target_compile_options(StateLazy01 PRIVATE /W4 /permissive- /O2)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo)
if(ipo)
  set_property(TARGET StateLazy01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()

enable_testing()
add_test(NAME state_lazy_01_model COMMAND StateLazy01 --self-test)
add_executable(StateLazy01Allocations main.cpp ${engine_objects})
target_compile_definitions(StateLazy01Allocations PRIVATE SB02_ALLOCATIONS=1 SB02_REQUEST_LIFETIME=1)
target_compile_features(StateLazy01Allocations PRIVATE cxx_std_20)
target_include_directories(StateLazy01Allocations PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(StateLazy01Allocations PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
target_compile_options(StateLazy01Allocations PRIVATE /W4 /permissive- /O2)
set_property(TARGET StateLazy01Allocations PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME state_lazy_01_allocation_model COMMAND StateLazy01Allocations --self-test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME state_lazy_01_oracle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_oracle.py" "$<TARGET_FILE:StateLazy01>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")

add_test(NAME state_lazy_01_empty_allocations COMMAND StateLazy01Allocations --container-probe)
