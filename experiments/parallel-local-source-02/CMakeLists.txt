cmake_minimum_required(VERSION 3.25)
project(ParallelLocalSource02 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(SERVER_ENGINE_BUILD_BENCHMARKS ON CACHE BOOL "Source list provider" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Only experiment tests" FORCE)
add_subdirectory("${ENGINE_ROOT}" engine EXCLUDE_FROM_ALL)
if(NOT SERVER_ENGINE_SOURCE_LEXER_TOKENS_V2)
 message(FATAL_ERROR "This slice requires ready Source lexer tokens")
endif()
get_target_property(sources ServerEngineV4PublishBenchmark SOURCES)
set(core)
foreach(source IN LISTS sources)
 if(NOT IS_ABSOLUTE "${source}")
  set(source "${ENGINE_ROOT}/${source}")
 endif()
 if(NOT source MATCHES "publish_benchmark[.]cpp$")
  list(APPEND core "${source}")
 endif()
endforeach()
add_executable(ParallelLocalSource02 main.cpp ${core})
target_compile_features(ParallelLocalSource02 PRIVATE cxx_std_20)
target_include_directories(ParallelLocalSource02 PRIVATE "${ENGINE_ROOT}/server_engine")
get_directory_property(definitions DIRECTORY "${ENGINE_ROOT}" COMPILE_DEFINITIONS)
target_compile_definitions(ParallelLocalSource02 PRIVATE ${definitions})
foreach(property COMPILE_OPTIONS LINK_LIBRARIES LINK_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE)
 get_target_property(value ServerEngineV4PublishBenchmark ${property})
 if(value AND NOT value MATCHES "NOTFOUND$")
  set_property(TARGET ParallelLocalSource02 PROPERTY ${property} "${value}")
 endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME parallel_local_source02_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py"
 "$<TARGET_FILE:ParallelLocalSource02>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
