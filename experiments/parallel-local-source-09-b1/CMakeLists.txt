cmake_minimum_required(VERSION 3.25)
project(IndexedReferenceState09B1 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-09" base EXCLUDE_FROM_ALL)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py reference_state.hpp)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(target PublishReferenceStateB1 PublishReferenceAuditB1 SourceReferenceSemanticB1 ReferenceStateTestsB1)
 if(target STREQUAL ReferenceStateTestsB1)
  add_executable(${target} state_tests.cpp)
 else()
  if(target STREQUAL PublishReferenceAuditB1)
   set(mode audit)
  else()
   set(mode native)
  endif()
  if(target STREQUAL SourceReferenceSemanticB1)
   set(mode native)
   add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/native/main.cpp")
   target_compile_definitions(${target} PRIVATE PLS03_PROBE=1)
  else()
   add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}/publish_parser.cpp")
  endif()
  target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}")
  get_target_property(libraries PublishFinalState08 LINK_LIBRARIES)
  target_link_libraries(${target} PRIVATE ${libraries})
  foreach(property COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
   get_target_property(value PublishFinalState08 ${property})
   if(value AND NOT value MATCHES "NOTFOUND$")
    set_property(TARGET ${target} PROPERTY ${property} "${value}")
   endif()
  endforeach()
 endif()
 target_compile_features(${target} PRIVATE cxx_std_20)
endforeach()
target_compile_definitions(PublishReferenceAuditB1 PRIVATE PLS09_DETAILS=1 PLS03_ALLOCATIONS=1)
# COMPILE_DEFINITIONS is copied above; append this selector afterwards.
target_compile_definitions(SourceReferenceSemanticB1 PRIVATE PLS03_PROBE=1)
enable_testing()
add_test(NAME B1_state COMMAND ReferenceStateTestsB1)
add_test(NAME B1_contexts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_contexts.py" "$<TARGET_FILE:SourceReferenceSemanticB1>" "${CMAKE_CURRENT_BINARY_DIR}/contexts")
foreach(mode 0 1)
 add_test(NAME B1_native_${mode} COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_resident.py" "$<TARGET_FILE:PublishReferenceStateB1>" "${ENGINE_ROOT}/build/parallel-local-source-08/writer-fixtures")
 set_tests_properties(B1_native_${mode} PROPERTIES ENVIRONMENT "PLS09_INDEXED=${mode}" RUN_SERIAL TRUE)
 foreach(suite source writer gaps)
  if(suite STREQUAL source)
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-02/test.py")
  elseif(suite STREQUAL writer)
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_writer.py")
  else()
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_gaps.py")
  endif()
  add_test(NAME B1_${suite}_${mode} COMMAND "${Python3_EXECUTABLE}" "${script}" "$<TARGET_FILE:SourceReferenceSemanticB1>" "${CMAKE_CURRENT_BINARY_DIR}/${suite}-${mode}")
  set_tests_properties(B1_${suite}_${mode} PROPERTIES ENVIRONMENT "PLS09_INDEXED=${mode};PLS03_MODE=1;PLS04_EARLY=1;PLS06_DENSE=1;PLS07_OWN=1;PLS07_VERIFY=1" RUN_SERIAL TRUE)
 endforeach()
endforeach()
