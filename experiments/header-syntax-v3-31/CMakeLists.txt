cmake_minimum_required(VERSION 3.25)
project(HeaderSyntaxV331 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED SERVERENGINE_ROOT OR NOT EXISTS "${SERVERENGINE_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "-DSERVERENGINE_ROOT=isolated/engine/overlay is required")
endif()
get_filename_component(SERVERENGINE_ROOT "${SERVERENGINE_ROOT}" ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SERVER_ENGINE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SERVERENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/engine")
get_target_property(_sources ServerEngineV4 SOURCES)
list(FILTER _sources EXCLUDE REGEX "server_engine/main\\.cpp$")
set(_absolute_sources)
foreach(_src IN LISTS _sources)
    if(IS_ABSOLUTE "${_src}")
        list(APPEND _absolute_sources "${_src}")
    else()
        list(APPEND _absolute_sources "${SERVERENGINE_ROOT}/${_src}")
    endif()
endforeach()
foreach(target IN ITEMS ServerEngineV4HeaderSyntaxV331Perf ServerEngineV4HeaderSyntaxV331Alloc)
    add_executable(${target} benchmarks/v3_31_perf_benchmark.cpp ${_absolute_sources})
    target_include_directories(${target} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks"
        "${SERVERENGINE_ROOT}"
        "${SERVERENGINE_ROOT}/server_engine")
    if(WIN32)
        target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
    else()
        find_package(Threads REQUIRED)
        target_link_libraries(${target} PRIVATE Threads::Threads)
        if(NOT APPLE)
            target_link_libraries(${target} PRIVATE rt)
        endif()
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
target_compile_definitions(ServerEngineV4HeaderSyntaxV331Alloc PRIVATE CW_V323_ALLOCATION_PROFILE=1)
