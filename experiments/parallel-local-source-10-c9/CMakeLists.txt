cmake_minimum_required(VERSION 3.25)
project(SourceCost10C9 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PLS10_CORE_LIBRARY "" CACHE FILEPATH "Matching Release IPO PLS08Core")
if(NOT PLS10_CORE_LIBRARY)
 message(FATAL_ERROR "Set PLS10_CORE_LIBRARY to the matching C7 baseline core")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(PublishCost10C9 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp")
target_compile_features(PublishCost10C9 PRIVATE cxx_std_20)
target_include_directories(PublishCost10C9 PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
target_compile_definitions(PublishCost10C9 PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
target_compile_options(PublishCost10C9 PRIVATE /W4 /permissive- /O2)
set_property(TARGET PublishCost10C9 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
target_link_libraries(PublishCost10C9 PRIVATE "${PLS10_CORE_LIBRARY}" psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
add_executable(SourceAudit10C9 EXCLUDE_FROM_ALL "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_graph.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_identity.cpp" allocation.cpp)
foreach(prop COMPILE_FEATURES INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_LIBRARIES)
 get_target_property(value PublishCost10C9 ${prop})
 set_property(TARGET SourceAudit10C9 PROPERTY ${prop} "${value}")
endforeach()
target_compile_definitions(SourceAudit10C9 PRIVATE PLS10_C9_AUDIT=1)
