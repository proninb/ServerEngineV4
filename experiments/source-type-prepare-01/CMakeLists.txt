cmake_minimum_required(VERSION 3.25)
project(SourceTypePrepare01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/publish-critical-path-02/symbols/ServerEngineV4PublishBenchmark.dir/Release" CACHE PATH "Complete current post-SHA Release+IPO objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
list(FILTER engine_objects EXCLUDE REGEX "[/\\](source_parser_v2|publish_benchmark)\.obj$")
if(NOT engine_objects)
 message(FATAL_ERROR "Build current production objects first")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
foreach(target SourceTypePrepare01 SourceTypePrepare01Audit SourceTypePrepare01Tests SourceTypePrepare01ProductionTests SourceTypePrepare01Native SourceTypePrepare01NativeTests)
 if(target MATCHES "Tests$")
  add_executable(${target} test.cpp "${ENGINE_ROOT}/server_engine/project/parser/source_parser_v2.cpp" ${engine_objects})
 else()
  add_executable(${target} "${ENGINE_ROOT}/benchmarks/publish_benchmark.cpp" "${ENGINE_ROOT}/server_engine/project/parser/source_parser_v2.cpp" ${engine_objects})
 endif()
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_definitions(${target} PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
 if(target STREQUAL "SourceTypePrepare01Native" OR target STREQUAL "SourceTypePrepare01NativeTests")
  target_compile_definitions(${target} PRIVATE CW_SOURCE_TYPE_PREPARE_V2=1)
 elseif(NOT target STREQUAL "SourceTypePrepare01ProductionTests")
  target_compile_definitions(${target} PRIVATE CW_SOURCE_TYPE_PREPARE_AB=1)
 endif()
 target_compile_options(${target} PRIVATE /O2 /Ob2 /DNDEBUG /Zi /W4 /permissive-)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine")
 target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
 target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
 set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endforeach()
target_compile_definitions(SourceTypePrepare01Audit PRIVATE CW_SOURCE_TYPE_PREPARE_AUDIT=1)
target_compile_definitions(SourceTypePrepare01Tests PRIVATE CW_SOURCE_TYPE_PREPARE_TEST=1)
enable_testing()
add_test(NAME source_type_prepare_reference COMMAND SourceTypePrepare01Tests)
set_tests_properties(source_type_prepare_reference PROPERTIES ENVIRONMENT "CW_SOURCE_TYPE_PREPARE=0")
add_test(NAME source_type_prepare_reuse COMMAND SourceTypePrepare01Tests)
set_tests_properties(source_type_prepare_reuse PROPERTIES ENVIRONMENT "CW_SOURCE_TYPE_PREPARE=1")
add_test(NAME source_type_prepare_production COMMAND SourceTypePrepare01ProductionTests)
add_test(NAME source_type_prepare_native_candidate COMMAND SourceTypePrepare01NativeTests)
add_test(NAME source_type_prepare_unavailable COMMAND SourceTypePrepare01Tests)
set_tests_properties(source_type_prepare_unavailable PROPERTIES ENVIRONMENT "CW_SOURCE_TYPE_PREPARE=1;CW_SOURCE_TYPE_PREPARE_UNAVAILABLE=1")
