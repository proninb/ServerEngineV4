cmake_minimum_required(VERSION 3.25)
project(PublishAliasAudit09 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-08" base EXCLUDE_FROM_ALL)
add_custom_target(PublishAliasBaseline09 DEPENDS PublishFinalState08)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py
 "${ENGINE_ROOT}/experiments/parallel-local-source-08/generate.py"
 "${ENGINE_ROOT}/experiments/parallel-local-source-08/publish_source.hpp"
 "${ENGINE_ROOT}/experiments/parallel-local-source-03/probe.hpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(target PublishAliasPhases09 PublishAliasDetails09)
 add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp")
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated")
 foreach(property COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
  get_target_property(value PublishFinalState08 ${property})
  if(value AND NOT value MATCHES "NOTFOUND$")
   set_property(TARGET ${target} PROPERTY ${property} "${value}")
  endif()
 endforeach()
 get_target_property(libraries PublishFinalState08 LINK_LIBRARIES)
 target_link_libraries(${target} PRIVATE ${libraries})
endforeach()
target_compile_definitions(PublishAliasDetails09 PRIVATE PLS09_DETAILS=1 PLS03_ALLOCATIONS=1)
enable_testing()
add_test(NAME PLS09_phase_resident COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_resident.py" "$<TARGET_FILE:PublishAliasPhases09>" "${ENGINE_ROOT}/build/parallel-local-source-08/writer-fixtures")
add_test(NAME PLS09_detail_resident COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_resident.py" "$<TARGET_FILE:PublishAliasDetails09>" "${ENGINE_ROOT}/build/parallel-local-source-08/writer-fixtures")
set_tests_properties(PLS09_phase_resident PLS09_detail_resident PROPERTIES RUN_SERIAL TRUE)
add_test(NAME PLS09_negative_semantics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_negative.py" "${CMAKE_CURRENT_BINARY_DIR}" "${ENGINE_ROOT}/build/parallel-local-source-08/fixtures")
set_tests_properties(PLS09_negative_semantics PROPERTIES RUN_SERIAL TRUE)
