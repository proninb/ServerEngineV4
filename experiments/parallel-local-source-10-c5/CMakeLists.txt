cmake_minimum_required(VERSION 3.25)
project(StreamingProjectSource10C5 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PLS10_CORE_LIBRARY "" CACHE FILEPATH "Optional already-built matching PLS08Core library")
if(PLS10_CORE_LIBRARY)
 add_library(PLS08Core STATIC IMPORTED)
 set_target_properties(PLS08Core PROPERTIES IMPORTED_LOCATION "${PLS10_CORE_LIBRARY}")
else()
 add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-08" base EXCLUDE_FROM_ALL)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3/commit.hpp" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3/publish_source.hpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(PublishStreamingSource10C5 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp")
target_compile_features(PublishStreamingSource10C5 PRIVATE cxx_std_20)
target_include_directories(PublishStreamingSource10C5 PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(PLS10_CORE_LIBRARY)
 target_compile_definitions(PublishStreamingSource10C5 PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
 target_compile_options(PublishStreamingSource10C5 PRIVATE /W4 /permissive- /O2)
 set_property(TARGET PublishStreamingSource10C5 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
 target_link_libraries(PublishStreamingSource10C5 PRIVATE PLS08Core psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
else()
 foreach(property LINK_LIBRARIES COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
  get_target_property(value PublishFinalState08 ${property})
  if(value AND NOT value MATCHES "NOTFOUND$")
   set_property(TARGET PublishStreamingSource10C5 PROPERTY ${property} "${value}")
  endif()
 endforeach()
endif()
enable_testing()
add_test(NAME PLS10C5_streaming COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py" "$<TARGET_FILE:PublishStreamingSource10C5>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
