cmake_minimum_required(VERSION 3.25)
project(PublishLifecycle01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/publish-no-sha256-01/ServerEngineV4PublishBenchmark.dir/Release" CACHE PATH "Complete current SHA-policy build objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
list(FILTER engine_objects EXCLUDE REGEX "[/\\](shm_type_batch|project_full_construction|publish_benchmark)\\.obj$")
if(NOT engine_objects)
 message(FATAL_ERROR "Build publish-no-sha256-01 benchmark first")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
foreach(target RuntimeRanges01 ConstructionLifetime01 LifecycleTests01)
 if(target STREQUAL "LifecycleTests01")
  add_executable(${target} test.cpp ${engine_objects})
  target_sources(${target} PRIVATE "${ENGINE_ROOT}/server_engine/project/runtime/runtime_layout.cpp" "${ENGINE_ROOT}/server_engine/project/runtime/fixed_direct_materializer.cpp")
 else()
  add_executable(${target} "${ENGINE_ROOT}/benchmarks/publish_benchmark.cpp" ${engine_objects})
 endif()
 target_sources(${target} PRIVATE "${ENGINE_ROOT}/server_engine/project/shm/shm_type_batch.cpp" "${ENGINE_ROOT}/server_engine/project/project_full_construction.cpp")
 target_compile_definitions(${target} PRIVATE CW_PUBLISH_NO_SHA256_AB=1)
 if(target STREQUAL "RuntimeRanges01" OR target STREQUAL "LifecycleTests01")
  target_compile_definitions(${target} PRIVATE CW_RUNTIME_API_RANGES_AB=1)
 endif()
 if(target STREQUAL "ConstructionLifetime01")
  target_compile_definitions(${target} PRIVATE CW_CONSTRUCTION_EARLY_RELEASE_AB=1)
 endif()
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_options(${target} PRIVATE /O2 /Ob2 /DNDEBUG /Zi /W4 /permissive-)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine")
 target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
 target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
 set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endforeach()
enable_testing()
add_test(NAME runtime_ranges_both COMMAND LifecycleTests01)

target_link_libraries(PublishHashAudit01 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
set_property(TARGET PublishHashAudit01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)

list(FILTER production_objects EXCLUDE REGEX "[/\\](file_context|project_full_construction|publish_benchmark)\\.obj$")
set_source_files_properties(${production_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)

 target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
