cmake_minimum_required(VERSION 3.25)
project(SourceResolution01 LANGUAGES CXX)
option(SOURCE_RESOLUTION_DIAGNOSTICS "Use diagnostic Parser proxies" ON)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(SERVER_ENGINE_BUILD_BENCHMARKS ON CACHE BOOL "Build benchmark" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Only own tests" FORCE)
add_subdirectory("${ENGINE_ROOT}" engine EXCLUDE_FROM_ALL)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py candidate.py capture.hpp
    "${ENGINE_ROOT}/server_engine/project/parser/source_parser_v2.cpp"
    "${ENGINE_ROOT}/server_engine/project/frontend/semantic_input.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
    "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(original_sources ServerEngineV4PublishBenchmark SOURCES)
set(bench_sources)
set(core_sources)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/candidate.py"
    "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(source IN LISTS original_sources)
    if(source MATCHES "semantic_input[.]cpp$")
        continue()
    endif()
    if(source MATCHES "[/\\]source_parser_v2\\.cpp$")
        continue()
    endif()
    if(NOT IS_ABSOLUTE "${source}")
        set(source "${ENGINE_ROOT}/${source}")
    endif()
    list(APPEND bench_sources "${source}")
    if(NOT source MATCHES "[/\\]publish_benchmark\\.cpp$")
        list(APPEND core_sources "${source}")
    endif()
endforeach()
if(SOURCE_RESOLUTION_DIAGNOSTICS)
    set(parser_source "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp")
else()
    set(parser_source "${ENGINE_ROOT}/server_engine/project/parser/source_parser_v2.cpp")
endif()
list(APPEND bench_sources "${parser_source}")
list(APPEND bench_sources "${CMAKE_CURRENT_BINARY_DIR}/generated/semantic_input.cpp")
list(APPEND core_sources "${CMAKE_CURRENT_BINARY_DIR}/generated/semantic_input.cpp")
set_property(TARGET ServerEngineV4PublishBenchmark PROPERTY SOURCES "${bench_sources}")
set_target_properties(ServerEngineV4PublishBenchmark PROPERTIES OUTPUT_NAME SourceResolution01)
add_executable(SourceResolution01Tests "${ENGINE_ROOT}/tests/frontend_regression_tests.cpp"
    "${parser_source}" ${core_sources})
target_compile_features(SourceResolution01Tests PRIVATE cxx_std_20)
target_include_directories(SourceResolution01Tests PRIVATE "${ENGINE_ROOT}/server_engine")
get_directory_property(engine_definitions DIRECTORY "${ENGINE_ROOT}" COMPILE_DEFINITIONS)
target_compile_definitions(SourceResolution01Tests PRIVATE ${engine_definitions})
foreach(property COMPILE_OPTIONS LINK_LIBRARIES LINK_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE)
    get_target_property(value ServerEngineV4PublishBenchmark ${property})
    if(value AND NOT value MATCHES "NOTFOUND$")
        set_property(TARGET SourceResolution01Tests PROPERTY ${property} "${value}")
    endif()
endforeach()
add_custom_target(SourceResolution01 DEPENDS ServerEngineV4PublishBenchmark SourceResolution01Tests)
enable_testing()
add_test(NAME source_resolution01_frontend COMMAND SourceResolution01Tests)
set_tests_properties(source_resolution01_frontend PROPERTIES ENVIRONMENT "CW_SOURCE_RESOLUTION_MODE=0")
add_test(NAME source_resolution01_ready_cursor COMMAND SourceResolution01Tests)
set_tests_properties(source_resolution01_ready_cursor PROPERTIES ENVIRONMENT "CW_SOURCE_RESOLUTION_MODE=0;CW_SOURCE_READY_CURSOR=1")
if(SOURCE_RESOLUTION_DIAGNOSTICS)
add_test(NAME source_resolution01_oracles COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/validate.py" "$<TARGET_FILE:ServerEngineV4PublishBenchmark>"
    "${CMAKE_CURRENT_BINARY_DIR}/validation")
endif()
