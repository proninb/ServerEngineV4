cmake_minimum_required(VERSION 3.25)
project(SingleBindingLookup10C8 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PLS10_CORE_LIBRARY "" CACHE FILEPATH "Optional matching prebuilt PLS08Core")
if(PLS10_CORE_LIBRARY)
 add_library(PLS08Core STATIC IMPORTED)
 set_target_properties(PLS08Core PROPERTIES IMPORTED_LOCATION "${PLS10_CORE_LIBRARY}")
else()
 add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-08" base EXCLUDE_FROM_ALL)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py selector.hpp "${ENGINE_ROOT}/server_engine/project/graph/graph.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(target PublishSingleLookup10C8 SingleLookupAudit10C8 SingleLookupTests10C8)
 if(target STREQUAL SingleLookupTests10C8)
  add_executable(${target} link_tests.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/single_graph.cpp")
 else()
  add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/single_graph.cpp")
 endif()
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${ENGINE_ROOT}/server_engine/project/graph" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
 if(PLS10_CORE_LIBRARY)
  target_compile_definitions(${target} PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
  target_compile_options(${target} PRIVATE /W4 /permissive- /O2)
  set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  target_link_libraries(${target} PRIVATE PLS08Core psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
 else()
  foreach(property LINK_LIBRARIES COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
   get_target_property(value PublishFinalState08 ${property})
   if(value AND NOT value MATCHES "NOTFOUND$")
    set_property(TARGET ${target} PROPERTY ${property} "${value}")
   endif()
  endforeach()
 endif()
endforeach()
target_compile_definitions(SingleLookupAudit10C8 PRIVATE PLS10_C8_AUDIT=1)
enable_testing()
add_test(NAME PLS10C8_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py" "$<TARGET_FILE:PublishSingleLookup10C8>" "$<TARGET_FILE:SingleLookupTests10C8>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
