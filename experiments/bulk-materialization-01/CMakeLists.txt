cmake_minimum_required(VERSION 3.25)
project(BulkMaterialization01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(SERVER_ENGINE_BUILD_BENCHMARKS ON CACHE BOOL "Benchmark" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Own tests" FORCE)
add_subdirectory("${ENGINE_ROOT}" engine EXCLUDE_FROM_ALL)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py bulk.hpp
 "${ENGINE_ROOT}/server_engine/project/graph/graph.cpp" "${ENGINE_ROOT}/server_engine/project/parser/source_parser_v2.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
 "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(sources ServerEngineV4PublishBenchmark SOURCES)
set(core)
foreach(source IN LISTS sources)
 if(NOT IS_ABSOLUTE "${source}")
  set(source "${ENGINE_ROOT}/${source}")
 endif()
 if(source MATCHES "[/\\]publish_benchmark[.]cpp$")
  set(main "${source}")
 elseif(NOT source MATCHES "[/\\]graph[.]cpp$|[/\\]source_parser_v2[.]cpp$")
  list(APPEND core "${source}")
 endif()
endforeach()
list(APPEND core "${CMAKE_CURRENT_BINARY_DIR}/generated/graph.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp")
add_library(BulkMaterialization01Core STATIC ${core})
add_executable(BulkMaterialization01Tests "${ENGINE_ROOT}/tests/frontend_regression_tests.cpp")
add_executable(BulkMaterialization01PrefixTests prefix_tests.cpp)
get_directory_property(definitions DIRECTORY "${ENGINE_ROOT}" COMPILE_DEFINITIONS)
foreach(target BulkMaterialization01Core BulkMaterialization01Tests BulkMaterialization01PrefixTests)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine")
 target_compile_definitions(${target} PRIVATE ${definitions} CW_RESOLVED_G_01=1)
 foreach(property COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE)
  get_target_property(value ServerEngineV4PublishBenchmark ${property})
  if(value AND NOT value MATCHES "NOTFOUND$")
   set_property(TARGET ${target} PROPERTY ${property} "${value}")
  endif()
 endforeach()
endforeach()
set_property(TARGET ServerEngineV4PublishBenchmark PROPERTY SOURCES "${main}")
target_compile_definitions(ServerEngineV4PublishBenchmark PRIVATE CW_RESOLVED_G_01=1)
set_target_properties(ServerEngineV4PublishBenchmark PROPERTIES OUTPUT_NAME BulkMaterialization01)
get_target_property(libs ServerEngineV4PublishBenchmark LINK_LIBRARIES)
get_target_property(options ServerEngineV4PublishBenchmark LINK_OPTIONS)
if(libs AND NOT libs MATCHES "NOTFOUND$")
 target_link_libraries(BulkMaterialization01Tests PRIVATE ${libs})
 target_link_libraries(BulkMaterialization01PrefixTests PRIVATE ${libs})
endif()
if(options AND NOT options MATCHES "NOTFOUND$")
 target_link_options(BulkMaterialization01Tests PRIVATE ${options})
endif()
target_link_libraries(ServerEngineV4PublishBenchmark PRIVATE BulkMaterialization01Core)
target_link_libraries(BulkMaterialization01Tests PRIVATE BulkMaterialization01Core)
target_link_libraries(BulkMaterialization01PrefixTests PRIVATE BulkMaterialization01Core)
add_custom_target(BulkMaterialization01 DEPENDS ServerEngineV4PublishBenchmark BulkMaterialization01Tests BulkMaterialization01PrefixTests)
enable_testing()
add_test(NAME bulk01_artifacts_and_errors COMMAND "${Python3_EXECUTABLE}"
 "${CMAKE_CURRENT_SOURCE_DIR}/validate.py" "$<TARGET_FILE:ServerEngineV4PublishBenchmark>" "${CMAKE_CURRENT_BINARY_DIR}/validation")
add_test(NAME bulk01_frontend_control COMMAND BulkMaterialization01Tests)
set_tests_properties(bulk01_frontend_control PROPERTIES ENVIRONMENT "CW_BULK_MODE=0")
foreach(block 1 7 65536)
 add_test(NAME bulk01_frontend_${block} COMMAND BulkMaterialization01Tests)
 set_tests_properties(bulk01_frontend_${block} PROPERTIES ENVIRONMENT "CW_BULK_MODE=1;CW_BULK_BLOCK=${block}")
 add_test(NAME bulk01_prefix_${block} COMMAND BulkMaterialization01PrefixTests)
 set_tests_properties(bulk01_prefix_${block} PROPERTIES ENVIRONMENT "CW_BULK_MODE=1;CW_BULK_BLOCK=${block}")
endforeach()
