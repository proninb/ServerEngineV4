cmake_minimum_required(VERSION 3.25)
project(FinalBindingReduction10A LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-09-b1" base EXCLUDE_FROM_ALL)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py binding_body.inc binding_reduction.hpp)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(target PublishFinalBinding10A FinalBindingAudit10A FinalBindingSemantic10A BindingReductionTests10A)
 if(target STREQUAL BindingReductionTests10A)
  add_executable(${target} state_tests.cpp)
 else()
  if(target STREQUAL FinalBindingAudit10A)
   set(mode audit)
  else()
   set(mode native)
  endif()
  if(target STREQUAL FinalBindingSemantic10A)
   add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/native/main.cpp")
  else()
   add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}/publish_parser.cpp")
  endif()
  target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}")
  get_target_property(libraries PublishFinalState08 LINK_LIBRARIES)
  target_link_libraries(${target} PRIVATE ${libraries})
  foreach(property COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
   get_target_property(value PublishFinalState08 ${property})
   if(value AND NOT value MATCHES "NOTFOUND$")
    set_property(TARGET ${target} PROPERTY ${property} "${value}")
   endif()
  endforeach()
 endif()
 target_compile_features(${target} PRIVATE cxx_std_20)
endforeach()
target_compile_definitions(FinalBindingAudit10A PRIVATE PLS09_DETAILS=1 PLS03_ALLOCATIONS=1)
target_compile_definitions(FinalBindingSemantic10A PRIVATE PLS03_PROBE=1)
enable_testing()
add_test(NAME PLS10_state COMMAND BindingReductionTests10A)
add_test(NAME PLS10_dependencies COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_dependencies.py" "$<TARGET_FILE:FinalBindingSemantic10A>" "${CMAKE_CURRENT_BINARY_DIR}/dependencies")
foreach(mode 0 1 2)
 add_test(NAME PLS10_contexts_${mode} COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-09-b1/test_contexts.py" "$<TARGET_FILE:FinalBindingSemantic10A>" "${CMAKE_CURRENT_BINARY_DIR}/contexts-${mode}")
 set_tests_properties(PLS10_contexts_${mode} PROPERTIES ENVIRONMENT "PLS10_FINALIZE=${mode}")
 add_test(NAME PLS10_native_${mode} COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_resident.py" "$<TARGET_FILE:PublishFinalBinding10A>" "${ENGINE_ROOT}/build/parallel-local-source-08/writer-fixtures")
 set_tests_properties(PLS10_native_${mode} PROPERTIES ENVIRONMENT "PLS09_INDEXED=1;PLS10_FINALIZE=${mode}" RUN_SERIAL TRUE)
 foreach(suite source writer gaps)
  if(suite STREQUAL source)
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-02/test.py")
  elseif(suite STREQUAL writer)
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_writer.py")
  else()
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_gaps.py")
  endif()
  add_test(NAME PLS10_${suite}_${mode} COMMAND "${Python3_EXECUTABLE}" "${script}" "$<TARGET_FILE:FinalBindingSemantic10A>" "${CMAKE_CURRENT_BINARY_DIR}/${suite}-${mode}")
  set_tests_properties(PLS10_${suite}_${mode} PROPERTIES ENVIRONMENT "PLS10_FINALIZE=${mode};PLS09_INDEXED=1;PLS03_MODE=1;PLS04_EARLY=1;PLS06_DENSE=1;PLS07_OWN=1;PLS07_VERIFY=1" RUN_SERIAL TRUE)
 endforeach()
endforeach()
