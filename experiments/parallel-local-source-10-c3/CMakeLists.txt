cmake_minimum_required(VERSION 3.25)
project(OrderedSource10C3 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-08" base EXCLUDE_FROM_ALL)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py commit.hpp publish_source.hpp stats.hpp)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(PublishOrderedSource10C3 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp")
target_include_directories(PublishOrderedSource10C3 PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(PublishOrderedSource10C3 PRIVATE cxx_std_20)
foreach(property LINK_LIBRARIES COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
 get_target_property(value PublishFinalState08 ${property})
 if(value AND NOT value MATCHES "NOTFOUND$")
  set_property(TARGET PublishOrderedSource10C3 PROPERTY ${property} "${value}")
 endif()
endforeach()
enable_testing()
add_test(NAME PLS10C3_ordered COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py" "$<TARGET_FILE:PublishOrderedSource10C3>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
