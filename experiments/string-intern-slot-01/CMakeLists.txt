cmake_minimum_required(VERSION 3.25)
project(StringInternSlot01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/source-cpu-sampling-02/current-v2/ServerEngineV4PublishBenchmark.dir/Release" CACHE PATH "Current production Release+IPO objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
list(FILTER engine_objects EXCLUDE REGEX "[/\\](string_table|publish_benchmark)\\.obj$")
if(NOT engine_objects)
  message(FATAL_ERROR "Build current-v2 sampling executable first")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
foreach(target StringInternSlot01 StringInternSlot01Tests)
  if(target STREQUAL "StringInternSlot01Tests")
    add_executable(${target} test.cpp "${ENGINE_ROOT}/server_engine/project/string/string_table.cpp" ${engine_objects})
    target_sources(${target} PRIVATE "${ENGINE_ROOT}/server_engine/project/runtime/runtime_layout.cpp" "${ENGINE_ROOT}/server_engine/project/runtime/fixed_direct_materializer.cpp")
  else()
    add_executable(${target} "${ENGINE_ROOT}/benchmarks/publish_benchmark.cpp" "${ENGINE_ROOT}/server_engine/project/string/string_table.cpp" ${engine_objects})
  endif()
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE CW_STRING_INTERN_SLOT_AB=1)
  target_compile_options(${target} PRIVATE /O2 /Ob2 /DNDEBUG /Zi /W4 /permissive-)
  target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine")
  target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
  target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
  set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endforeach()
enable_testing()
add_test(NAME string_intern_slot_01 COMMAND StringInternSlot01Tests)
