cmake_minimum_required(VERSION 3.25)
project(PLS10C2 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py slot_commit.hpp "${ENGINE_ROOT}/experiments/parallel-local-source-10-c1/feasibility.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/c1_fixtures.inc" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-10-c1" c1 EXCLUDE_FROM_ALL)
get_target_property(sources DirectCommitFeasibility10C1 SOURCES)
list(FILTER sources EXCLUDE REGEX "^feasibility.cpp$")
add_executable(SlotCommit10C2 tests.cpp ${sources})
target_compile_features(SlotCommit10C2 PRIVATE cxx_std_20)
target_include_directories(SlotCommit10C2 PRIVATE "${ENGINE_ROOT}/server_engine" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
 target_compile_options(SlotCommit10C2 PRIVATE /W4 /permissive-)
 set_property(TARGET SlotCommit10C2 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
enable_testing()
add_test(NAME PLS10C2_slots COMMAND SlotCommit10C2)
