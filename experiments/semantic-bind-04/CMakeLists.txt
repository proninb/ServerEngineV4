cmake_minimum_required(VERSION 3.25)
project(SemanticBind04 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/semantic-bind-03/SemanticBind03Engine.dir/Release" CACHE PATH "Previously built MSVC Release engine objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
if(NOT engine_objects)
  message(FATAL_ERROR "Build SemanticBind03Engine first and pass its Release object directory")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(SemanticBind04 main.cpp ${engine_objects})
target_compile_features(SemanticBind04 PRIVATE cxx_std_20)
target_include_directories(SemanticBind04 PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(SemanticBind04 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
if(MSVC)
  target_compile_options(SemanticBind04 PRIVATE /W4 /permissive- /O2)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo)
if(ipo)
  set_property(TARGET SemanticBind04 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()

enable_testing()
add_test(NAME semantic_bind_04_model COMMAND SemanticBind04 --self-test)

add_executable(SemanticBind04Legacy main.cpp ${engine_objects})
target_compile_definitions(SemanticBind04Legacy PRIVATE SB04_LEGACY=1)
target_compile_features(SemanticBind04Legacy PRIVATE cxx_std_20)
target_include_directories(SemanticBind04Legacy PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(SemanticBind04Legacy PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
target_compile_options(SemanticBind04Legacy PRIVATE /W4 /permissive- /O2)
set_property(TARGET SemanticBind04Legacy PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME semantic_bind_04_legacy COMMAND SemanticBind04Legacy --self-test)

add_executable(SemanticBind04Allocations main.cpp ${engine_objects})
target_compile_definitions(SemanticBind04Allocations PRIVATE SB02_ALLOCATIONS=1 SB02_REQUEST_LIFETIME=1)
target_compile_features(SemanticBind04Allocations PRIVATE cxx_std_20)
target_include_directories(SemanticBind04Allocations PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(SemanticBind04Allocations PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
target_compile_options(SemanticBind04Allocations PRIVATE /W4 /permissive- /O2)
set_property(TARGET SemanticBind04Allocations PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME semantic_bind_04_allocation_model COMMAND SemanticBind04Allocations --self-test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME semantic_bind_04_oracle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_oracle.py" "$<TARGET_FILE:SemanticBind04>" "$<TARGET_FILE:SemanticBind04Legacy>" "$<TARGET_FILE:SemanticBind04Allocations>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
