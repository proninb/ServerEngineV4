cmake_minimum_required(VERSION 3.25)
project(SourceCostBreakdown01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(OBJECT_DIR "${ENGINE_ROOT}/build/publish-end-to-end-profile-01/ServerEngineV4PublishBenchmark.dir/Release" CACHE PATH "Fresh default production benchmark objects")
file(GLOB objects "${OBJECT_DIR}/*.obj")
if(NOT objects)
  message(FATAL_ERROR "Build the root default Release ServerEngineV4PublishBenchmark first")
endif()
list(FILTER objects EXCLUDE REGEX "/source_parser_v2\\.obj$")
set_source_files_properties(${objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py
  "${ENGINE_ROOT}/server_engine/project/parser/source_parser_v2.cpp"
  "${ENGINE_ROOT}/server_engine/project/parser/source_endpoint_resolver.hpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(SourceCostBreakdown01 "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp" ${objects})
target_compile_features(SourceCostBreakdown01 PRIVATE cxx_std_20)
target_include_directories(SourceCostBreakdown01 PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_compile_definitions(SourceCostBreakdown01 PRIVATE CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1 CW_SOURCE_V2_STRING_PATH_REUSE_01=1)
target_compile_options(SourceCostBreakdown01 PRIVATE /W4 /permissive- /O2)
target_link_libraries(SourceCostBreakdown01 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
set_property(TARGET SourceCostBreakdown01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
enable_testing()
add_executable(SourceCostBreakdown01Tests test_cost.cpp)
target_compile_features(SourceCostBreakdown01Tests PRIVATE cxx_std_20)
target_compile_options(SourceCostBreakdown01Tests PRIVATE /UNDEBUG)
add_test(NAME source_cost_01_accounting COMMAND SourceCostBreakdown01Tests)
add_test(NAME source_cost_01_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/validate.py"
  "${ENGINE_ROOT}/build/publish-end-to-end-profile-01/Release/ServerEngineV4PublishBenchmark.exe"
  "$<TARGET_FILE:SourceCostBreakdown01>" "${CMAKE_CURRENT_BINARY_DIR}/validation")
