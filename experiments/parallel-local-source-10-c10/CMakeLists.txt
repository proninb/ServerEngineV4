cmake_minimum_required(VERSION 3.25)
project(SourceScratch10C10 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PLS10_CORE_LIBRARY "" CACHE FILEPATH "Matching C7 Release IPO core")
if(NOT PLS10_CORE_LIBRARY)
 message(FATAL_ERROR "Set PLS10_CORE_LIBRARY")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(SourceAllocationAudit10C10 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_graph.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_identity.cpp" allocation.cpp)
target_compile_features(SourceAllocationAudit10C10 PRIVATE cxx_std_20)
target_include_directories(SourceAllocationAudit10C10 PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
target_compile_definitions(SourceAllocationAudit10C10 PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1 PLS10_C9_AUDIT=1)
target_compile_options(SourceAllocationAudit10C10 PRIVATE /W4 /permissive- /O2)
set_property(TARGET SourceAllocationAudit10C10 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
target_link_libraries(SourceAllocationAudit10C10 PRIVATE "${PLS10_CORE_LIBRARY}" psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
add_executable(PublishScratch10C10 "${CMAKE_CURRENT_BINARY_DIR}/generated/native/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/native/publish_parser.cpp")
target_compile_features(PublishScratch10C10 PRIVATE cxx_std_20)
target_include_directories(PublishScratch10C10 PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated/native" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
target_compile_definitions(PublishScratch10C10 PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
target_compile_options(PublishScratch10C10 PRIVATE /W4 /permissive- /O2)
set_property(TARGET PublishScratch10C10 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
target_link_libraries(PublishScratch10C10 PRIVATE "${PLS10_CORE_LIBRARY}" psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py selector.hpp)
enable_testing()
add_test(NAME PLS10C10_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.py" "$<TARGET_FILE:PublishScratch10C10>" "$<TARGET_FILE:SourceAllocationAudit10C10>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
