cmake_minimum_required(VERSION 3.25)
project(SparseIndirectWrites10B LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_subdirectory("${ENGINE_ROOT}/experiments/parallel-local-source-10-a" base EXCLUDE_FROM_ALL)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py reduce_body.inc registry.hpp write_store.hpp selector.hpp)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(target PublishSparseWrites10B SparseWritesAudit10B SparseWritesSemantic10B SparseWriteTests10B)
 if(target STREQUAL SparseWriteTests10B)
  add_executable(${target} state_tests.cpp)
  target_compile_definitions(${target} PRIVATE PLS10B_PROBE_HEADER="${ENGINE_ROOT}/experiments/parallel-local-source-03/probe.hpp")
 else()
  if(target STREQUAL SparseWritesAudit10B)
   set(mode audit)
  else()
   set(mode native)
  endif()
  if(target STREQUAL SparseWritesSemantic10B)
   add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/native/main.cpp")
  else()
   add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}/publish_parser.cpp")
  endif()
  target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${CMAKE_CURRENT_BINARY_DIR}/generated/${mode}")
  get_target_property(libraries PublishFinalState08 LINK_LIBRARIES)
  target_link_libraries(${target} PRIVATE ${libraries})
  foreach(property COMPILE_DEFINITIONS COMPILE_OPTIONS INTERPROCEDURAL_OPTIMIZATION_RELEASE LINK_OPTIONS)
   get_target_property(value PublishFinalState08 ${property})
   if(value AND NOT value MATCHES "NOTFOUND$")
    set_property(TARGET ${target} PROPERTY ${property} "${value}")
   endif()
  endforeach()
 endif()
 target_compile_features(${target} PRIVATE cxx_std_20)
endforeach()
target_compile_definitions(SparseWritesAudit10B PRIVATE PLS09_DETAILS=1 PLS03_ALLOCATIONS=1 PLS10B_PROBE_HEADER="${CMAKE_CURRENT_BINARY_DIR}/generated/audit/probe.hpp")
target_compile_definitions(SparseWritesSemantic10B PRIVATE PLS03_PROBE=1)
enable_testing()
add_test(NAME PLS10B_state COMMAND SparseWriteTests10B)
add_test(NAME PLS10B_merge COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_merge.py" "$<TARGET_FILE:SparseWritesSemantic10B>" "${CMAKE_CURRENT_BINARY_DIR}/merge")
add_test(NAME PLS10B_dependencies COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-10-a/test_dependencies.py" "$<TARGET_FILE:SparseWritesSemantic10B>" "${CMAKE_CURRENT_BINARY_DIR}/dependencies")
set_tests_properties(PLS10B_dependencies PROPERTIES ENVIRONMENT "PLS10_WRITES=1")
foreach(mode 0 1)
 add_test(NAME PLS10B_contexts_${mode} COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-09-b1/test_contexts.py" "$<TARGET_FILE:SparseWritesSemantic10B>" "${CMAKE_CURRENT_BINARY_DIR}/contexts-${mode}")
 set_tests_properties(PLS10B_contexts_${mode} PROPERTIES ENVIRONMENT "PLS10_FINALIZE=2;PLS10_WRITES=${mode}")
 add_test(NAME PLS10B_native_${mode} COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_resident.py" "$<TARGET_FILE:PublishSparseWrites10B>" "${ENGINE_ROOT}/build/parallel-local-source-08/writer-fixtures")
 set_tests_properties(PLS10B_native_${mode} PROPERTIES ENVIRONMENT "PLS09_INDEXED=1;PLS10_FINALIZE=2;PLS10_WRITES=${mode}" RUN_SERIAL TRUE)
 foreach(suite source writer gaps)
  if(suite STREQUAL source)
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-02/test.py")
  elseif(suite STREQUAL writer)
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_writer.py")
  else()
   set(script "${ENGINE_ROOT}/experiments/parallel-local-source-08/test_gaps.py")
  endif()
  add_test(NAME PLS10B_${suite}_${mode} COMMAND "${Python3_EXECUTABLE}" "${script}" "$<TARGET_FILE:SparseWritesSemantic10B>" "${CMAKE_CURRENT_BINARY_DIR}/${suite}-${mode}")
  set_tests_properties(PLS10B_${suite}_${mode} PROPERTIES ENVIRONMENT "PLS10_FINALIZE=2;PLS10_WRITES=${mode};PLS09_INDEXED=1;PLS03_MODE=1;PLS04_EARLY=1;PLS06_DENSE=1;PLS07_OWN=1;PLS07_VERIFY=1" RUN_SERIAL TRUE)
 endforeach()
endforeach()
