cmake_minimum_required(VERSION 3.25)
project(HeaderSyntaxV343 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED SERVERENGINE_ROOT OR NOT EXISTS "${SERVERENGINE_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "-DSERVERENGINE_ROOT=isolated-overlay is required")
endif()
get_filename_component(SERVERENGINE_ROOT "${SERVERENGINE_ROOT}" ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SERVER_ENGINE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SERVERENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/engine")
get_target_property(_sources ServerEngineV4 SOURCES)
list(FILTER _sources EXCLUDE REGEX "server_engine/main\\.cpp$")
set(_absolute_sources)
foreach(_src IN LISTS _sources)
    if(IS_ABSOLUTE "${_src}")
        list(APPEND _absolute_sources "${_src}")
    else()
        list(APPEND _absolute_sources "${SERVERENGINE_ROOT}/${_src}")
    endif()
endforeach()
if(NOT WIN32)
    find_package(Threads REQUIRED)
endif()
foreach(kind IN ITEMS Perf Alloc)
    set(target "ServerEngineV4HeaderSyntaxV343${kind}")
    add_executable(${target} "benchmarks/v3_42_perf_benchmark.cpp" ${_absolute_sources})
    target_include_directories(${target} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks" "${SERVERENGINE_ROOT}" "${SERVERENGINE_ROOT}/server_engine")
    if(WIN32)
        target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
    else()
        target_link_libraries(${target} PRIVATE Threads::Threads)
        if(NOT APPLE)
            target_link_libraries(${target} PRIVATE rt)
        endif()
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(kind STREQUAL "Alloc")
        target_compile_definitions(${target} PRIVATE CW_V323_ALLOCATION_PROFILE=1)
    endif()
endforeach()
# Explicit Release-safe tests from V3-37R, reused unmodified.
enable_testing()
add_executable(V343RangeTests tests/test_v337r_ranges.cpp)
add_executable(V343ScannerTests tests/test_v337r_scanner_nested.cpp)
foreach(t IN ITEMS V343RangeTests V343ScannerTests)
    target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks" "${SERVERENGINE_ROOT}")
    target_compile_features(${t} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
add_test(NAME V343RangeTests COMMAND V343RangeTests)
add_test(NAME V343ScannerTests COMMAND V343ScannerTests)

add_executable(V343TypeSpecTests tests/test_v342_type_spec.cpp)
target_include_directories(V343TypeSpecTests PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_project"
    "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks" "${SERVERENGINE_ROOT}")
target_compile_features(V343TypeSpecTests PRIVATE cxx_std_20)
add_test(NAME V343TypeSpecTests COMMAND V343TypeSpecTests)
