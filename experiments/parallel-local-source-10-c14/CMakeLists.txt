cmake_minimum_required(VERSION 3.25)
project(RemainingCommitAttribution10C14 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PLS10_CORE_LIBRARY "" CACHE FILEPATH "Matching C10 MSVC Release IPO core")
if(NOT PLS10_CORE_LIBRARY)
 message(FATAL_ERROR "Set PLS10_CORE_LIBRARY")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(PublishCommitProfile10C14 "${CMAKE_CURRENT_BINARY_DIR}/generated/native/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/native/publish_parser.cpp")
add_executable(CommitAllocationAudit10C14 "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/publish_parser.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/source_parser_v2.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_graph.cpp" "${CMAKE_CURRENT_BINARY_DIR}/generated/audit_identity.cpp" allocation.cpp)
foreach(target PublishCommitProfile10C14 CommitAllocationAudit10C14)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_include_directories(${target} PRIVATE "${ENGINE_ROOT}/server_engine" "${ENGINE_ROOT}/server_engine/project/parser" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c3")
 target_compile_definitions(${target} PRIVATE CW_SOURCE_V2_STRING_PATH_REUSE_01=1 CW_SOURCE_PREHASH_PARALLEL_V2=1 CW_SOURCE_LEXER_TOKENS_V2=1 CW_SOURCE_STRING_BATCH_V2=1)
 target_compile_options(${target} PRIVATE /W4 /permissive- /O2)
 set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
 target_link_libraries(${target} PRIVATE "${PLS10_CORE_LIBRARY}" psapi bcrypt ws2_32 crypt32 ncrypt advapi32 winhttp)
endforeach()
target_include_directories(PublishCommitProfile10C14 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/native")
target_include_directories(CommitAllocationAudit10C14 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(CommitAllocationAudit10C14 PRIVATE PLS10_C9_AUDIT=1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS generate.py probe.hpp)
add_executable(CommitClockCalibration10C14 calibration.cpp)
target_compile_features(CommitClockCalibration10C14 PRIVATE cxx_std_20)
target_compile_options(CommitClockCalibration10C14 PRIVATE /W4 /permissive- /O2)
set_property(TARGET CommitClockCalibration10C14 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
enable_testing()
add_test(NAME PLS10C14_parity COMMAND "${Python3_EXECUTABLE}" "${ENGINE_ROOT}/experiments/parallel-local-source-10-c10/test.py" "$<TARGET_FILE:PublishCommitProfile10C14>" "$<TARGET_FILE:CommitAllocationAudit10C14>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
set_tests_properties(PLS10C14_parity PROPERTIES ENVIRONMENT "PLS10_C14_PROFILE=1")
