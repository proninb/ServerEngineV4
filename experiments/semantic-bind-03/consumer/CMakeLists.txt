cmake_minimum_required(VERSION 3.25)
project(SemanticBind03Consumer LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/semantic-bind-03/SemanticBind03Engine.dir/Release" CACHE PATH "Previously built MSVC Release engine objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
if(NOT engine_objects)
  message(FATAL_ERROR "Build SemanticBind03Engine first and pass its Release object directory")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(SemanticBind03Consumer main.cpp ${engine_objects})
target_compile_features(SemanticBind03Consumer PRIVATE cxx_std_20)
target_include_directories(SemanticBind03Consumer PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(SemanticBind03Consumer PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
if(MSVC)
  target_compile_options(SemanticBind03Consumer PRIVATE /W4 /permissive- /O2)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo)
if(ipo)
  set_property(TARGET SemanticBind03Consumer PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
