cmake_minimum_required(VERSION 3.25)
project(SemanticBind03 LANGUAGES CXX)
include(CheckIPOSupported)
check_ipo_supported(RESULT sb03_ipo)
add_subdirectory("../semantic-bind-01" sb01)
get_target_property(full_sources SemanticBind01 SOURCES)
list(FILTER full_sources EXCLUDE REGEX "(^|/)main\\.cpp$")
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
add_library(SemanticBind03Engine OBJECT ${full_sources})
set(targets SemanticBind03Engine)
foreach(kind IN ITEMS Timing Allocations)
  set(target "SemanticBind03${kind}")
  add_executable(${target} main.cpp $<TARGET_OBJECTS:SemanticBind03Engine>)
  list(APPEND targets ${target})
  if(WIN32)
    target_link_libraries(${target} PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(${target} PRIVATE Threads::Threads)
  endif()
  if(kind STREQUAL "Allocations")
    target_compile_definitions(${target} PRIVATE SB02_ALLOCATIONS=1 SB02_REQUEST_LIFETIME=1)
  endif()
endforeach()
foreach(target IN LISTS targets)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /O2)
  endif()
  if(sb03_ipo)
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
endforeach()
enable_testing()
add_test(NAME compact_model COMMAND SemanticBind03Timing --self-test)
add_test(NAME compact_allocations COMMAND SemanticBind03Allocations --self-test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME compact_oracle_lifetimes COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/test_variants.py" "$<TARGET_FILE:SemanticBind03Timing>"
  "$<TARGET_FILE:SemanticBind03Allocations>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
