cmake_minimum_required(VERSION 3.25)
project(HeaderChunk01 LANGUAGES CXX)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
set(ENGINE_OBJECT_DIR "${ENGINE_ROOT}/build/semantic-bind-03/SemanticBind03Engine.dir/Release" CACHE PATH "Previously built MSVC Release engine objects")
file(GLOB engine_objects "${ENGINE_OBJECT_DIR}/*.obj")
if(NOT engine_objects)
  message(FATAL_ERROR "Build SemanticBind03Engine first and pass its Release object directory")
endif()
set_source_files_properties(${engine_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(HeaderChunk01 main.cpp ${engine_objects})
target_compile_features(HeaderChunk01 PRIVATE cxx_std_20)
target_include_directories(HeaderChunk01 PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(HeaderChunk01 PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
if(MSVC)
  target_compile_options(HeaderChunk01 PRIVATE /W4 /permissive- /O2)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo)
if(ipo)
  set_property(TARGET HeaderChunk01 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()

enable_testing()
add_test(NAME header_chunk_boundaries COMMAND HeaderChunk01 --self-test)
add_executable(HeaderChunk01Allocations main.cpp ${engine_objects})
target_compile_definitions(HeaderChunk01Allocations PRIVATE SB02_ALLOCATIONS=1 SB02_REQUEST_LIFETIME=1)
target_compile_features(HeaderChunk01Allocations PRIVATE cxx_std_20)
target_include_directories(HeaderChunk01Allocations PRIVATE "${ENGINE_ROOT}" "${ENGINE_ROOT}/server_engine")
target_link_libraries(HeaderChunk01Allocations PRIVATE Psapi bcrypt crypt32 ncrypt winhttp ws2_32)
target_compile_options(HeaderChunk01Allocations PRIVATE /W4 /permissive- /O2)
set_property(TARGET HeaderChunk01Allocations PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME header_chunk_allocation_model COMMAND HeaderChunk01Allocations --self-test)
